function getRecommendation(sessions, exercise) {
if (sessions.length === 0) return null;
const last = sessions[sessions.length - 1];
if (!last.sets || last.sets.length === 0) return null;

const weights = last.sets.map((s) => parseFloat(s.weight)).filter(Boolean);
if (weights.length === 0) return null;
const maxWeight = Math.max(…weights);
const avgReps = last.sets.reduce((a, s) => a + (parseInt(s.reps) || 0), 0) / last.sets.length;

let increment = 2.5;
if (LOWER_BODY.has(exercise)) increment = 5;
if (OLYMPIC.has(exercise)) increment = 2.5;
if (exercise.includes(“curl”) || exercise.includes(“Curl”) || exercise.includes(“Triceps”) || exercise.includes(“Face pull”) || exercise.includes(“Élévation”)) increment = 1.25;

let advice = “”;
let nextWeight = maxWeight;
let color = “#FBBF24”;

if (avgReps >= 8 && sessions.length >= 1) {
nextWeight = maxWeight + increment;
advice = `Toutes les reps complétées facilement → augmente de ${increment}kg. Objectif : ${nextWeight}kg.`;
color = “#C8FF00”;
} else if (avgReps >= 6) {
nextWeight = maxWeight;
advice = `Bonne séance. Maintiens ${maxWeight}kg et cherche à finir toutes les séries proprement avant d'augmenter.`;
color = “#FBBF24”;
} else {
nextWeight = Math.max(0, maxWeight - increment);
advice = `Reps insuffisantes (< 6 en moyenne). Réduis à ${nextWeight}kg et consolide la technique.`;
color = “#F472B6”;
}

// Compare with previous session
if (sessions.length >= 2) {
const prev = sessions[sessions.length - 2];
const prevWeights = (prev.sets || []).map((s) => parseFloat(s.weight)).filter(Boolean);
if (prevWeights.length > 0) {
const prevMax = Math.max(…prevWeights);
const diff = maxWeight - prevMax;
if (diff > 0) advice += ` +${diff}kg vs séance précédente. 📈`;
else if (diff < 0) advice += ` ${diff}kg vs séance précédente. Surveille la fatigue.`;
}
}

return { nextWeight, advice, color, lastWeight: maxWeight };
}
