const epley = (w, r) => r === 1 ? w : Math.round(w * (1 + r / 30));
const brzycki = (w, r) => r >= 37 ? w : Math.round(w * 36 / (37 - r));

const pctOf = (oneRM, pct) => Math.round(oneRM * pct / 100 / 2.5) * 2.5;

// ─── REST TIMER ───────────────────────────────────────────────────────────────
function RestTimer() {
  const [preset, setPreset] = useState(90);
  const [rem, setRem] = useState(90);
  const [active, setActive] = useState(false);
  const iRef = useRef(null);
  useEffect(() => {
    if (active) { iRef.current = setInterval(() => setRem((r) => { if (r <= 1) { clearInterval(iRef.current); setActive(false); return preset; } return r - 1; }), 1000); }
    else clearInterval(iRef.current);
    return () => clearInterval(iRef.current);
  }, [active, preset]);
  const fmt = (s) => `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
  const pct = (preset - rem) / preset;
  const r = 28, cx = 34, cy = 34, circ = 2 * Math.PI * r;
  return (
    <div style={{ ...c.card(), background: active ? "#0A1800" : T.card, border: `1px solid ${active ? T.accent + "55" : T.border}`, transition: "all 0.3s" }}>
      <div style={c.title}>⏱ TIMER DE REPOS</div>
      <div style={{ display: "flex", alignItems: "center", gap: 16 }}>
        <div style={{ position: "relative", width: 68, height: 68, flexShrink: 0 }}>
          <svg width={68} height={68} style={{ transform: "rotate(-90deg)" }}>
            <circle cx={cx} cy={cy} r={r} fill="none" stroke={T.dim} strokeWidth={5} />
            <circle cx={cx} cy={cy} r={r} fill="none" stroke={active ? T.accent : T.muted} strokeWidth={5}
              strokeDasharray={circ} strokeDashoffset={circ * (1 - pct)}
              strokeLinecap="round" style={{ transition: "stroke-dashoffset 0.9s linear, stroke 0.3s" }} />
          </svg>
          <div style={{ position: "absolute", inset: 0, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 15, fontWeight: 700, color: active ? T.accent : T.muted }}>{fmt(rem)}</div>
        </div>
        <div style={{ flex: 1 }}>
          <div style={{ display: "flex", gap: 5, marginBottom: 8 }}>
            {[60, 90, 120, 180].map((p) => (
              <button key={p} onClick={() => { setPreset(p); setRem(p); setActive(false); }}
                style={{ flex: 1, background: preset === p ? T.accent : T.dim, color: preset === p ? T.bg : T.muted, border: "none", borderRadius: 5, padding: "4px 0", fontSize: 10, fontWeight: 700, fontFamily: "inherit", cursor: "pointer" }}>
                {fmt(p)}
              </button>
            ))}
          </div>
          <div style={{ display: "flex", gap: 6 }}>
            <button onClick={() => { if (!active) setRem(preset); setActive(!active); }}
              style={{ flex: 2, background: active ? T.combat + "30" : T.accent, color: active ? T.combat : T.bg, border: active ? `1px solid ${T.combat + "50"}` : "none", borderRadius: 6, padding: "7px", fontSize: 11, fontWeight: 700, fontFamily: "inherit", cursor: "pointer" }}>
              {active ? "⏸ PAUSE" : "▶ START"}
            </button>
            <button onClick={() => { setActive(false); setRem(preset); }}
              style={{ flex: 1, background: T.dim, color: T.muted, border: "none", borderRadius: 6, padding: "7px", fontSize: 10, fontFamily: "inherit", cursor: "pointer" }}>RESET</button>
          </div>
        </div>
      </div>
    </div>
  );
}

// ─── HOME TAB ────────────────────────────────────────────────────────────────
function HomeTab({ sessions, bodyData, recovery, onTab }) {
  const today = new Date().toISOString().split("T")[0];
  const weekAgo = new Date(Date.now() - 7 * 864e5).toISOString().split("T")[0];
  const thisWeek = sessions.filter((s) => s.dateISO >= weekAgo);
  const todayLogged = sessions.some((s) => s.dateISO === today);

  // Streak
  let streak = 0;
  const days = [...new Set(sessions.map((s) => s.dateISO))].sort().reverse();
  let check = today;
  for (const d of days) {
    const diff = (new Date(check) - new Date(d)) / 864e5;
    if (diff <= 1) { streak++; check = d; } else break;
  }

  // Deload alert
  const lastRecov = recovery.slice(-3);
  const avgFatigue = lastRecov.length >= 3 ? lastRecov.reduce((a, r) => a + r.fatigue, 0) / 3 : 0;
  const avgSleep = lastRecov.filter((r) => r.sleep).length >= 2 ? lastRecov.filter((r) => r.sleep).reduce((a, r) => a + r.sleep, 0) / lastRecov.filter((r) => r.sleep).length : 8;
  const recentRPE = sessions.slice(-5).flatMap((s) => (s.sets || []).filter((x) => x.rpe).map((x) => +x.rpe));
  const avgRPE = recentRPE.length ? recentRPE.reduce((a, b) => a + b, 0) / recentRPE.length : 0;
  const needsDeload = avgFatigue >= 4 || avgSleep < 6.5 || avgRPE >= 8.8;

  // PRs
  const prMap = {};
  sessions.forEach((s) => {
    (s.sets || []).forEach((x) => {
      if (!x.weight || !x.reps) return;
      const orm = est1RM(+x.weight, +x.reps);
      if (!prMap[s.exercise] || orm > prMap[s.exercise].orm) prMap[s.exercise] = { orm, date: s.date, w: x.weight, r: x.reps };
    });
  });
  const prs = Object.entries(prMap).sort((a, b) => b[1].date > a[1].date ? 1 : -1).slice(0, 5);

  // Weekly volume
  const weekVol = thisWeek.reduce((a, s) => a + (s.sets || []).reduce((b, x) => b + (+x.weight || 0) * (+x.reps || 0), 0), 0);

  const lastWeight = bodyData.length ? bodyData[bodyData.length - 1].weight : null;
  const prevWeight = bodyData.length >= 2 ? bodyData[bodyData.length - 2].weight : null;
  const weightDelta = lastWeight && prevWeight ? (lastWeight - prevWeight).toFixed(1) : null;

  return (
    <>
      {/* Greeting */}
      <div style={{ marginBottom: 14 }}>
        <div style={{ fontSize: 11, color: T.muted, letterSpacing: "1px", marginBottom: 4 }}>{new Date().toLocaleDateString("fr-FR", { weekday: "long", day: "numeric", month: "long" }).toUpperCase()}</div>
        <div style={{ fontSize: 20, fontWeight: 700, color: T.white, letterSpacing: "-0.5px" }}>
          {todayLogged ? "Séance loggée ✓" : "Quelle séance aujourd'hui ?"}
        </div>
      </div>

      {/* Deload alert */}
      {needsDeload && (
        <div style={{ ...c.accentCard(T.warn), color: "#FCD34D" }}>
          <div style={{ fontWeight: 700, marginBottom: 4, fontSize: 11 }}>⚠️ SIGNAL DE DÉCHARGE DÉTECTÉ</div>
          <div style={{ fontSize: 11, lineHeight: 1.6, color: "#FCD34D99" }}>
            {avgFatigue >= 4 ? `Fatigue élevée (moy. ${avgFatigue.toFixed(1)}/5). ` : ""}
            {avgSleep < 6.5 ? `Sommeil insuffisant (moy. ${avgSleep.toFixed(1)}h). ` : ""}
            {avgRPE >= 8.8 ? `RPE moyen élevé (${avgRPE.toFixed(1)}/10). ` : ""}
            → Réduis le volume de 40–50% cette semaine.
          </div>
        </div>
      )}

      {/* Stats */}
      <div style={c.grid3}>
        {[
          { val: streak, label: "Jours consécutifs", col: T.accent, icon: "🔥" },
          { val: thisWeek.length, label: "Séances / semaine", col: T.endurance, icon: "💪" },
          { val: prs.length, label: "PRs enregistrés", col: T.hyper, icon: "🏆" },
        ].map((st) => (
          <div key={st.label} style={{ background: T.card, border: `1px solid ${T.border}`, borderRadius: 10, padding: "12px 8px", textAlign: "center" }}>
            <div style={{ fontSize: 22, marginBottom: 2 }}>{st.icon}</div>
            <div style={{ fontSize: 22, fontWeight: 700, color: st.col }}>{st.val}</div>
            <div style={{ fontSize: 9, color: T.muted, letterSpacing: "0.5px", textTransform: "uppercase", marginTop: 2 }}>{st.label}</div>
          </div>
        ))}
      </div>

      {/* Quick stats row */}
      <div style={c.grid2}>
        <div style={{ background: T.card, border: `1px solid ${T.border}`, borderRadius: 10, padding: 12 }}>
          <div style={{ fontSize: 9, color: T.muted, letterSpacing: "1px", textTransform: "uppercase", marginBottom: 4 }}>VOLUME SEMAINE</div>
          <div style={{ fontSize: 18, fontWeight: 700, color: T.strength }}>{weekVol > 0 ? `${Math.round(weekVol / 1000)}t` : "—"}</div>
          <div style={{ fontSize: 9, color: T.muted }}>kg × reps total</div>
        </div>
        <div style={{ background: T.card, border: `1px solid ${T.border}`, borderRadius: 10, padding: 12 }}>
          <div style={{ fontSize: 9, color: T.muted, letterSpacing: "1px", textTransform: "uppercase", marginBottom: 4 }}>POIDS DE CORPS</div>
          <div style={{ fontSize: 18, fontWeight: 700, color: T.endurance }}>{lastWeight ? `${lastWeight}kg` : "—"}</div>
          {weightDelta && <div style={{ fontSize: 9, color: +weightDelta > 0 ? T.warn : T.power }}>{+weightDelta > 0 ? "+" : ""}{weightDelta}kg vs hier</div>}
        </div>
      </div>

      {/* Quick actions */}
      <div style={c.card()}>
        <div style={c.title}>ACCÈS RAPIDE</div>
        <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
          {[
            { label: "🎯 Choisir ma séance du jour", tab: "seance", col: T.accent },
            { label: "📓 Logger une séance", tab: "journal", col: T.strength },
            { label: "📊 Saisir mon suivi quotidien", tab: "suivi", col: T.endurance },
            { label: "🏆 Voir mes PRs", tab: "prs", col: T.hyper },
          ].map((a) => (
            <button key={a.tab} onClick={() => onTab(a.tab)}
              style={{ background: a.col + "15", color: a.col, border: `1px solid ${a.col + "30"}`, borderRadius: 8, padding: "10px 14px", fontSize: 11, fontWeight: 700, fontFamily: "inherit", cursor: "pointer", textAlign: "left" }}>
              {a.label}
            </button>
          ))}
        </div>
      </div>

      {/* Recent PRs */}
      {prs.length > 0 && (
        <div style={c.card()}>
          <div style={c.title}>🏆 DERNIERS PRs</div>
          {prs.map(([ex, pr], i) => (
            <div key={ex} style={{ ...c.row, borderBottom: i < prs.length - 1 ? `1px solid ${T.border}` : "none" }}>
              <div>
                <div style={{ fontSize: 11, color: T.white, fontWeight: 700 }}>{ex}</div>
                <div style={{ fontSize: 10, color: T.muted }}>{pr.date} · {pr.w}kg × {pr.r} reps</div>
              </div>
              <div style={{ textAlign: "right" }}>
                <div style={{ fontSize: 14, fontWeight: 700, color: T.hyper }}>≈{pr.orm}kg</div>
                <div style={{ fontSize: 9, color: T.muted }}>1RM est.</div>
              </div>
            </div>
          ))}
        </div>
      )}

      {/* Weekly calendar */}
      <div style={c.card()}>
        <div style={c.title}>📅 SEMAINE EN COURS</div>
        <div style={{ display: "grid", gridTemplateColumns: "repeat(7,1fr)", gap: 5 }}>
          {["L","M","M","J","V","S","D"].map((day, i) => {
            const date = new Date(); date.setDate(date.getDate() - date.getDay() + i + 1);
            const iso = date.toISOString().split("T")[0];
            const sessionsOnDay = sessions.filter((s) => s.dateISO === iso);
            const isToday = iso === today;
            const hasSesh = sessionsOnDay.length > 0;
            return (
              <div key={i} style={{ textAlign: "center" }}>
                <div style={{ fontSize: 9, color: T.muted, marginBottom: 4, fontWeight: isToday ? 700 : 400, color: isToday ? T.accent : T.muted }}>{day}</div>
                <div style={{ height: 32, borderRadius: 6, background: hasSesh ? T.accent : isToday ? T.dim : T.card2, border: `1px solid ${isToday ? T.accent + "50" : T.border}`, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 12 }}>
                  {hasSesh ? "✓" : isToday ? "·" : ""}
                </div>
                <div style={{ fontSize: 8, color: hasSesh ? T.accent : T.muted, marginTop: 2 }}>{sessionsOnDay.length ? `×${sessionsOnDay.length}` : ""}</div>
              </div>
            );
          })}
        </div>
      </div>
    </>
  );
}

// ─── JOURNAL TAB ─────────────────────────────────────────────────────────────
function JournalTab({ sessions, setSessions }) {
  const [view, setView] = useState("log");
  const [selEx, setSelEx] = useState(ALL_EX[0]);
  const [sets, setSets] = useState([{ reps: "", weight: "", rpe: "" }]);
  const [msg, setMsg] = useState("");
  const [histEx, setHistEx] = useState(ALL_EX[0]);
  const [chartMode, setChartMode] = useState("max");
  const [prFlash, setPrFlash] = useState(null);

  const persist = (data) => store.set("journal_v4", data);

  const getInc = (ex) => {
    if (LOWER_EX.has(ex)) return 5;
    if (["Power Clean","Hang Power Clean","KB Swing","KB Snatch"].includes(ex)) return 2.5;
    if (ex.includes("Curl") || ex.includes("Triceps") || ex.includes("Élévation") || ex.includes("Face pull")) return 1.25;
    return 2.5;
  };

  const logSession = async () => {
    const valid = sets.filter((s) => s.reps || s.weight);
    if (!valid.length) return;
    const entry = { id: Date.now(), date: new Date().toLocaleDateString("fr-FR"), dateISO: new Date().toISOString().split("T")[0], exercise: selEx, sets: valid };
    const updated = [...sessions, entry];

    // PR detection
    const prevSessions = sessions.filter((s) => s.exercise === selEx);
    const prevMaxOrm = Math.max(0, ...prevSessions.flatMap((s) => (s.sets || []).filter((x) => x.weight && x.reps).map((x) => est1RM(+x.weight, +x.reps))));
    const newMaxOrm = Math.max(0, ...valid.filter((x) => x.weight && x.reps).map((x) => est1RM(+x.weight, +x.reps)));
    if (newMaxOrm > prevMaxOrm && newMaxOrm > 0) setPrFlash({ ex: selEx, orm: newMaxOrm });

    setSessions(updated); await persist(updated);
    setSets([{ reps: "", weight: "", rpe: "" }]);
    setMsg("✓ Séance enregistrée !"); setTimeout(() => setMsg(""), 2500);
    setTimeout(() => setPrFlash(null), 4000);
  };

  const delSession = async (id) => { const u = sessions.filter((s) => s.id !== id); setSessions(u); await persist(u); };

  const histSessions = sessions.filter((s) => s.exercise === histEx).sort((a, b) => a.dateISO > b.dateISO ? 1 : -1);
  const chartData = histSessions.map((s) => {
    const ws = (s.sets || []).map((x) => parseFloat(x.weight)).filter(Boolean);
    const vol = (s.sets || []).reduce((a, x) => a + (+x.weight || 0) * (+x.reps || 0), 0);
    const rpeArr = (s.sets || []).filter((x) => x.rpe).map((x) => +x.rpe);
    return { date: s.date, max: ws.length ? Math.max(...ws) : 0, orm: ws.length ? Math.max(...(s.sets || []).filter((x) => x.weight && x.reps).map((x) => est1RM(+x.weight, +x.reps))) : 0, volume: Math.round(vol), rpe: rpeArr.length ? +(rpeArr.reduce((a, b) => a + b, 0) / rpeArr.length).toFixed(1) : null };
  });

  // Recommendation
  const getRec = () => {
    if (!histSessions.length) return null;
    const last = histSessions[histSessions.length - 1];
    const ws = (last.sets || []).map((x) => parseFloat(x.weight)).filter(Boolean);
    if (!ws.length) return null;
    const maxW = Math.max(...ws);
    const avgReps = (last.sets || []).reduce((a, s) => a + (+s.reps || 0), 0) / last.sets.length;
    const rpeArr = (last.sets || []).filter((s) => s.rpe).map((s) => +s.rpe);
    const avgRpe = rpeArr.length ? rpeArr.reduce((a, b) => a + b, 0) / rpeArr.length : null;
    const inc = getInc(histEx);
    let nextW = maxW, col = T.cardio, icon = "→", advice = "";
    if (avgReps >= 7 && (!avgRpe || avgRpe <= 8)) { nextW = maxW + inc; col = T.accent; icon = "📈"; advice = `Séance réussie → +${inc}kg. Vise ${nextW}kg.`; }
    else if (avgReps >= 5) { col = T.cardio; icon = "→"; advice = `Consolide ${maxW}kg avant d'augmenter.`; }
    else { nextW = Math.max(0, maxW - inc); col = T.hyper; icon = "📉"; advice = `Reps insuffisantes → réduis à ${nextW}kg.`; }
    if (avgRpe) advice += ` RPE moy. : ${avgRpe.toFixed(1)}/10.`;
    if (histSessions.length >= 2) {
      const prev = histSessions[histSessions.length - 2];
      const prevWs = (prev.sets || []).map((x) => parseFloat(x.weight)).filter(Boolean);
      if (prevWs.length) { const diff = maxW - Math.max(...prevWs); if (diff > 0) advice += ` +${diff}kg vs précédente.`; }
    }
    // 1RM
    const allOrm = (last.sets || []).filter((x) => x.weight && x.reps).map((x) => est1RM(+x.weight, +x.reps));
    const oneRM = allOrm.length ? Math.max(...allOrm) : null;
    return { nextW, col, icon, advice, oneRM };
  };
  const rec = getRec();
  const usesW = WEIGHT_EX.has(histEx);

  const recentForEx = sessions.filter((s) => s.exercise === selEx).slice(-3).reverse();

  const views = [{ id: "log", l: "📝 LOG" }, { id: "history", l: "📊 HISTORIQUE" }];

  return (
    <>
      <RestTimer />

      {prFlash && (
        <div style={{ background: T.hyper + "20", border: `1px solid ${T.hyper}60`, borderRadius: 10, padding: "12px 14px", marginBottom: 10, textAlign: "center" }}>
          <div style={{ fontSize: 22, marginBottom: 4 }}>🏆</div>
          <div style={{ fontSize: 13, fontWeight: 700, color: T.hyper }}>NOUVEAU PR — {prFlash.ex}</div>
          <div style={{ fontSize: 11, color: T.text, marginTop: 4 }}>1RM estimé : <strong style={{ color: T.hyper }}>{prFlash.orm}kg</strong></div>
        </div>
      )}

      <div style={{ display: "flex", gap: 6, marginBottom: 12 }}>
        {views.map((v) => <button key={v.id} onClick={() => setView(v.id)} style={{ flex: 1, background: view === v.id ? T.accent : T.card, color: view === v.id ? T.bg : T.muted, border: `1px solid ${view === v.id ? T.accent : T.border}`, borderRadius: 8, padding: 9, fontSize: 11, fontWeight: 700, fontFamily: "inherit", cursor: "pointer" }}>{v.l}</button>)}
      </div>

      {view === "log" && (
        <>
          <div style={c.card()}>
            <span style={c.label()}>EXERCICE</span>
            <select value={selEx} onChange={(e) => setSelEx(e.target.value)} style={c.sel}>
              {EX.map((g) => <optgroup key={g.g} label={g.g}>{g.list.map((ex) => <option key={ex} value={ex}>{ex}</option>)}</optgroup>)}
            </select>
            <span style={c.label()}>SÉRIES</span>
            <div style={{ display: "grid", gridTemplateColumns: "22px 1fr 1fr 1fr 28px", gap: 4, marginBottom: 8 }}>
              {["#","REPS","KG","RPE",""].map((h) => <span key={h} style={{ fontSize: 9, color: T.muted, textAlign: "center" }}>{h}</span>)}
            </div>
            {sets.map((sv, i) => (
              <div key={i} style={{ display: "grid", gridTemplateColumns: "22px 1fr 1fr 1fr 28px", gap: 4, marginBottom: 4 }}>
                <div style={{ background: T.dim, borderRadius: 4, height: 36, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 10, color: T.muted }}>{i + 1}</div>
                <input type="number" min="0" placeholder="10" value={sv.reps} onChange={(e) => setSets(sets.map((s, j) => j === i ? { ...s, reps: e.target.value } : s))} style={c.inp()} />
                <input type="number" min="0" step="0.5" placeholder="60" value={sv.weight} onChange={(e) => setSets(sets.map((s, j) => j === i ? { ...s, weight: e.target.value } : s))} style={c.inp(T.accent)} />
                <input type="number" min="1" max="10" placeholder="7" value={sv.rpe} onChange={(e) => setSets(sets.map((s, j) => j === i ? { ...s, rpe: e.target.value } : s))} style={c.inp(T.cardio)} />
                <button onClick={() => setSets(sets.filter((_, j) => j !== i))} style={{ background: "#1A0A0A", color: T.combat, border: `1px solid #3F1010`, borderRadius: 4, height: 36, width: 28, cursor: "pointer", fontSize: 14 }}>×</button>
              </div>
            ))}
            <div style={{ fontSize: 9, color: T.muted, textAlign: "right", marginBottom: 8 }}>RPE: 6=facile · 8=dur · 10=max</div>
            <button onClick={() => setSets([...sets, { reps: "", weight: "", rpe: "" }])} style={{ width: "100%", background: T.card2, color: T.muted, border: `1px dashed ${T.border}`, borderRadius: 6, padding: 8, fontSize: 10, fontFamily: "inherit", cursor: "pointer" }}>+ AJOUTER SÉRIE</button>
          </div>
          <button onClick={logSession} style={c.btn(T.accent, T.bg, true)}>ENREGISTRER LA SÉANCE</button>
          {msg && <div style={{ ...c.accentCard(T.accent), textAlign: "center", marginTop: 8 }}>{msg}</div>}

          {recentForEx.length > 0 && (
            <div style={{ ...c.card(), marginTop: 10 }}>
              <div style={c.title}>DERNIÈRES — {selEx.slice(0, 22).toUpperCase()}</div>
              {recentForEx.map((session, idx) => {
                const mw = Math.max(0, ...(session.sets || []).map((x) => +x.weight || 0));
                return (
                  <div key={session.id} style={{ borderBottom: idx < recentForEx.length - 1 ? `1px solid ${T.border}` : "none", paddingBottom: 10, marginBottom: 10 }}>
                    <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 6 }}>
                      <span style={{ fontSize: 11, color: T.muted }}>{session.date}</span>
                      <div style={{ display: "flex", gap: 8, alignItems: "center" }}>
                        {mw > 0 && <span style={{ fontSize: 11, color: T.accent, fontWeight: 700 }}>{mw}kg</span>}
                        <button onClick={() => delSession(session.id)} style={{ background: "none", border: "none", color: T.muted, cursor: "pointer", fontSize: 12 }}>🗑</button>
                      </div>
                    </div>
                    <div style={{ display: "flex", gap: 4, flexWrap: "wrap" }}>
                      {(session.sets || []).map((sv, si) => <span key={si} style={{ background: T.card2, color: T.white, fontSize: 10, padding: "2px 7px", borderRadius: 4 }}>{sv.reps}×{sv.weight}{sv.weight ? "kg" : ""}{sv.rpe ? ` @${sv.rpe}` : ""}</span>)}
                    </div>
                  </div>
                );
              })}
            </div>
          )}
        </>
      )}

      {view === "history" && (
        <>
          <div style={c.card()}>
            <span style={c.label()}>EXERCICE</span>
            <select value={histEx} onChange={(e) => setHistEx(e.target.value)} style={c.sel}>
              {EX.map((g) => <optgroup key={g.g} label={g.g}>{g.list.map((ex) => { const cnt = sessions.filter((s) => s.exercise === ex).length; return <option key={ex} value={ex}>{ex}{cnt ? ` (${cnt})` : ""}</option>; })}</optgroup>)}
            </select>
          </div>

          {histSessions.length === 0 ? (
            <div style={{ textAlign: "center", padding: "30px 16px", color: T.muted }}>
              <div style={{ fontSize: 32, marginBottom: 8 }}>📭</div>
              <div style={{ fontSize: 12 }}>Aucune séance enregistrée pour cet exercice.</div>
            </div>
          ) : (
            <>
              <div style={c.grid3}>
                {[{ val: histSessions.length, l: "Séances", col: T.accent }, { val: chartData.length ? Math.max(...chartData.map((d) => d.max)) + "kg" : "—", l: "Record kg", col: T.strength }, { val: chartData.length ? chartData[chartData.length - 1].max + "kg" : "—", l: "Dernière", col: T.endurance }].map((sv) => (
                  <div key={sv.l} style={{ background: T.card, border: `1px solid ${T.border}`, borderRadius: 10, padding: "10px 6px", textAlign: "center" }}>
                    <div style={{ fontSize: 16, fontWeight: 700, color: sv.col }}>{sv.val}</div>
                    <div style={{ fontSize: 9, color: T.muted, letterSpacing: "0.5px", textTransform: "uppercase", marginTop: 2 }}>{sv.l}</div>
                  </div>
                ))}
              </div>

              {rec && (
                <div style={{ ...c.accentCard(rec.col), marginBottom: 10 }}>
                  <div style={{ fontSize: 10, fontWeight: 700, color: rec.col, letterSpacing: "1px", marginBottom: 6 }}>🎯 RECOMMANDATION PROCHAINE SÉANCE</div>

                        {[50,60,70,75,80,85,90,95].map((p) => (
                          <span key={p} style={{ background: T.card2, border: `1px solid ${T.border}`, borderRadius: 5, padding: "3px 7px", fontSize: 10 }}>

T.bg : T.muted, border: "none", borderRadius: 5, padding: "4px 8px", fontSize: 9, fontWeight: 700, fontFamily: "inherit", cursor: "pointer" }}>{m.l}</button>

                      </defs>
                      <CartesianGrid strokeDasharray="3 3" stroke={T.border} />

T.hyper : T.cardio} strokeWidth={2} fill="url(#grad)" dot={{ fill: chartMode === "max" ? T.accent : chartMode === "orm" ? T.strength : chartMode === "volume" ? T.hyper : T.cardio, r: 3, strokeWidth: 0 }} connectNulls />

                })}
              </div>
            </>
          )}
        </>
      )}
    </>
  );
}

// ─── PRs TAB ─────────────────────────────────────────────────────────────────
function PRsTab({ sessions }) {
  const prMap = {};
  sessions.forEach((s) => {
    (s.sets || []).forEach((x) => {
      if (!x.weight || !x.reps) return;
      const orm = est1RM(+x.weight, +x.reps);
      if (!prMap[s.exercise] || orm > prMap[s.exercise].orm) prMap[s.exercise] = { orm, date: s.date, w: +x.weight, r: +x.reps };
    });
  });
  const prs = Object.entries(prMap).sort((a, b) => b[1].orm - a[1].orm);
  if (!prs.length) return <div style={{ textAlign: "center", padding: "40px 16px", color: T.muted }}><div style={{ fontSize: 40, marginBottom: 12 }}>🏆</div><div style={{ fontSize: 13 }}>Aucun PR encore enregistré.</div><div style={{ fontSize: 11, marginTop: 6, color: T.dim }}>Commence à logger tes séances pour voir tes PRs apparaître automatiquement.</div></div>;
  return (
    <>
      <div style={{ ...c.accentCard(T.hyper), marginBottom: 12 }}>
        <div style={{ fontSize: 12, fontWeight: 700, color: T.hyper, marginBottom: 4 }}>🏆 {prs.length} PERSONAL RECORDS</div>
        <div style={{ fontSize: 11, color: T.text }}>1RM estimé via formule Epley + Brzycki (moyenne). Mis à jour automatiquement à chaque séance.</div>
      </div>
      <div style={c.card()}>
        {prs.map(([ex, pr], i) => (
          <div key={ex} style={{ ...c.row, borderBottom: i < prs.length - 1 ? `1px solid ${T.border}` : "none", flexDirection: "column", alignItems: "stretch", gap: 6 }}>
            <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
              <div>
                <div style={{ fontSize: 12, color: T.white, fontWeight: 700 }}>{ex}</div>
                <div style={{ fontSize: 10, color: T.muted }}>{pr.date} · {pr.w}kg × {pr.r} rep{+pr.r > 1 ? "s" : ""}</div>
              </div>
              <div style={{ textAlign: "right" }}>
                <div style={{ fontSize: 18, fontWeight: 700, color: T.hyper }}>≈{pr.orm}kg</div>
                <div style={{ fontSize: 9, color: T.muted }}>1RM estimé</div>
              </div>
            </div>
            <div style={{ display: "flex", gap: 4, flexWrap: "wrap" }}>
              {[70, 75, 80, 85, 90, 95].map((p) => (
                <span key={p} style={{ background: T.card2, border: `1px solid ${T.border}`, borderRadius: 5, padding: "2px 7px", fontSize: 10 }}>
                  <span style={{ color: T.muted }}>{p}%=</span><span style={{ color: T.accent, fontWeight: 700 }}>{pctOf(pr.orm, p)}</span>
                </span>
              ))}
            </div>
          </div>
        ))}
      </div>
    </>
  );
}
