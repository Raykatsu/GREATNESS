function DayView({ days, showSci }) {
const [activeDay, setActiveDay] = useState(days[0].id);
const day = days.find((d) => d.id === activeDay);
return (
<>
<div style={{ display: “flex”, gap: “6px”, marginBottom: “16px”, flexWrap: “wrap” }}>
{days.map((d) => (
<button key={d.id} onClick={() => setActiveDay(d.id)} style={{ background: activeDay === d.id ? d.color : “#141720”, color: activeDay === d.id ? dark : muted, border: `1px solid ${activeDay === d.id ? d.color : cardBorder}`, borderRadius: “6px”, padding: “6px 12px”, fontSize: “10px”, fontWeight: “700”, fontFamily: “‘DM Mono’, monospace”, cursor: “pointer”, letterSpacing: “0.5px” }}>
{d.id.toUpperCase()}
</button>
))}
</div>
{day && (
<>
<div style={styles.dayChip(day.color)}>◆ {day.theme}</div>
<div style={{ display: “flex”, gap: “8px”, marginBottom: “16px” }}>
<span style={{ display: “inline-block”, background: “#1E2330”, color: muted, fontSize: “10px”, fontWeight: “700”, padding: “2px 8px”, borderRadius: “4px”, marginRight: “6px” }}>{day.duration}</span>
<span style={{ display: “inline-block”, background: day.color + “33”, color: day.color, fontSize: “10px”, fontWeight: “700”, padding: “2px 8px”, borderRadius: “4px” }}>INTENSITÉ {day.intensity.toUpperCase()}</span>
</div>
{day.blocks.map((block, bi) => (
<div key={bi} style={styles.card}>
<div style={styles.cardTitle}>{block.name}</div>
{block.exercises.map((ex, ei) => (
<div key={ei} style={{ …styles.row, borderBottom: ei < block.exercises.length - 1 ? `1px solid ${cardBorder}` : “none” }}>
<div style={styles.exo}>
<div style={{ fontSize: “12px”, color: white }}>{ex.name}</div>
{ex.note && <div style={{ fontSize: “10px”, color: muted, marginTop: “2px” }}>{ex.note}</div>}
</div>
<div style={styles.param}>{ex.params}</div>
</div>
))}
{showSci && block.sci && <div style={styles.sciNote}>🔬 {block.sci}</div>}
</div>
))}
</>
)}
</>
);
}
