const calcACWR=(sessions)=>{
  const now=new Date();
  const d7=new Date(now-7*864e5).toISOString().split("T")[0];
  const d28=new Date(now-28*864e5).toISOString().split("T")[0];
  const vol=(s)=>(s.sets||[]).reduce((a,x)=>a+(+x.kg||0)*(+x.reps||0)+(+x.duration_min||0)*10+(+x.rounds||0)*30,0);
  const acute=sessions.filter(s=>s.dateISO>=d7).reduce((a,s)=>a+vol(s),0)/7;
  const chronic=sessions.filter(s=>s.dateISO>=d28).reduce((a,s)=>a+vol(s),0)/28;
  if(!chronic)return null;
  return{ratio:+(acute/chronic).toFixed(2),acute:Math.round(acute),chronic:Math.round(chronic)};
};
