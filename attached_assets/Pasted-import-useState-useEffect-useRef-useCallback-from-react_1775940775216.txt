function matchesDuration(key, dur) {
  if (dur === "all") return true;
  const t = SESSION_TEMPLATES[key];
  const mins = parseInt(t.dur);
  if (dur === "court") return mins <= 45;
  if (dur === "long") return mins >= 60;
  return true;
}

function matchesFocus(key, foc) {
  if (foc === "all") return true;
  return key.startsWith(foc);
}

// ── REST TIMER ────────────────────────────────────────────────────────────────
function RestTimer() {
  const [active, setActive] = useState(false);
  const [preset, setPreset] = useState(90);
  const [remaining, setRemaining] = useState(90);
  const ref = useRef(null);

  useEffect(() => {
    if (active) {
      ref.current = setInterval(() => {
        setRemaining((r) => {
          if (r <= 1) { clearInterval(ref.current); setActive(false); return preset; }
          return r - 1;
        });
      }, 1000);
    } else clearInterval(ref.current);
    return () => clearInterval(ref.current);
  }, [active, preset]);

  const pct = ((preset - remaining) / preset) * 100;
  const fmt = (sec) => `${Math.floor(sec / 60)}:${String(sec % 60).padStart(2, "0")}`;

  return (
    <div style={{ background: C.card, border: `1px solid ${active ? C.accent + "77" : C.border}`, borderRadius: 8, padding: 14, marginBottom: 12 }}>
      <div style={{ fontSize: 11, fontWeight: 700, letterSpacing: 2, color: C.accent, marginBottom: 10 }}>⏱ TIMER DE REPOS</div>
      <div style={{ display: "flex", gap: 6, marginBottom: 12, flexWrap: "wrap" }}>
        {[60, 90, 120, 180].map((p) => (
          <button key={p} onClick={() => { setPreset(p); setRemaining(p); setActive(false); }} style={{ background: preset === p ? C.accent : "#1E2330", color: preset === p ? C.dark : C.muted, border: "none", borderRadius: 6, padding: "5px 12px", fontSize: 11, fontWeight: 700, fontFamily: "'DM Mono',monospace", cursor: "pointer" }}>
            {fmt(p)}
          </button>
        ))}
      </div>
      {/* Progress ring */}
      <div style={{ display: "flex", alignItems: "center", gap: 16 }}>
        <div style={{ position: "relative", width: 70, height: 70 }}>
          <svg width={70} height={70} style={{ transform: "rotate(-90deg)" }}>
            <circle cx={35} cy={35} r={30} fill="none" stroke={C.border} strokeWidth={5} />
            <circle cx={35} cy={35} r={30} fill="none" stroke={active ? C.accent : "#4B5563"} strokeWidth={5} strokeDasharray={`${2 * Math.PI * 30}`} strokeDashoffset={`${2 * Math.PI * 30 * (1 - pct / 100)}`} strokeLinecap="round" style={{ transition: "stroke-dashoffset 1s linear" }} />
          </svg>
          <div style={{ position: "absolute", inset: 0, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 16, fontWeight: 700, color: active ? C.accent : C.muted }}>{fmt(remaining)}</div>
        </div>
        <div style={{ flex: 1, display: "flex", flexDirection: "column", gap: 6 }}>
          <button onClick={() => { if (!active) { setRemaining(preset); } setActive(!active); }} style={{ background: active ? "#1A0A0A" : C.accent, color: active ? C.red : C.dark, border: active ? `1px solid ${C.red}` : "none", borderRadius: 6, padding: "8px", fontSize: 12, fontWeight: 700, fontFamily: "'DM Mono',monospace", cursor: "pointer" }}>
            {active ? "⏸ PAUSE" : "▶ START"}
          </button>
          <button onClick={() => { setActive(false); setRemaining(preset); }} style={{ background: "#1E2330", color: C.muted, border: "none", borderRadius: 6, padding: "5px", fontSize: 10, fontFamily: "'DM Mono',monospace", cursor: "pointer" }}>RESET</button>
        </div>
      </div>
    </div>
  );
}

// ── RECOMMENDATION ENGINE ─────────────────────────────────────────────────────
function getRecommendation(sessions, exercise) {
  if (!sessions.length) return null;
  const last = sessions[sessions.length - 1];
  if (!last.sets?.length) return null;
  const weights = last.sets.map((s) => parseFloat(s.weight)).filter(Boolean);
  if (!weights.length) return null;
  const maxW = Math.max(...weights);
  const avgReps = last.sets.reduce((a, s) => a + (parseInt(s.reps) || 0), 0) / last.sets.length;
  const avgRpe = last.sets.reduce((a, s) => a + (parseInt(s.rpe) || 0), 0) / last.sets.filter((s) => s.rpe).length;

  let inc = 2.5;
  if (LOWER_EX.has(exercise)) inc = 5;
  if (OLYMPIC_EX.has(exercise)) inc = 2.5;
  if (exercise.includes("Curl") || exercise.includes("Triceps") || exercise.includes("Élévation") || exercise.includes("Face pull")) inc = 1.25;

  let nextW = maxW;
  let advice = "";
  let color = C.yellow;
  let icon = "→";

  const rpeOk = isNaN(avgRpe) || avgRpe <= 8;
  const repsOk = avgReps >= 7;

  if (repsOk && rpeOk) {
    nextW = maxW + inc;
    advice = `Reps complètes + RPE dans la cible → augmente de ${inc}kg. Vise ${nextW}kg.`;
    color = C.accent; icon = "📈";
  } else if (repsOk && !rpeOk) {
    nextW = maxW;
    advice = `Reps ok mais RPE élevé (${avgRpe.toFixed(0)}/10) → maintiens ${maxW}kg. Consolide avant d'augmenter.`;
    color = C.yellow; icon = "⚠️";
  } else {
    nextW = Math.max(0, maxW - inc);
    advice = `Reps insuffisantes ou RPE trop haut → réduis à ${nextW}kg. Travaille la technique.`;
    color = C.pink; icon = "📉";
  }

  if (sessions.length >= 2) {
    const prev = sessions[sessions.length - 2];
    const prevMax = Math.max(...(prev.sets || []).map((s) => parseFloat(s.weight)).filter(Boolean));
    if (prevMax) {
      const diff = maxW - prevMax;
      if (diff > 0) advice += ` Progression : +${diff}kg vs séance précédente.`;
      else if (diff < 0) advice += ` Régression : ${diff}kg vs séance précédente — surveille la fatigue.`;
    }
  }

  return { nextW, advice, color, icon, lastW: maxW };
}

// ── JOURNAL TAB ───────────────────────────────────────────────────────────────
function JournalTab() {
  const [view, setView] = useState("log");
  const [sessions, setSessions] = useState([]);
  const [loaded, setLoaded] = useState(false);
  const [selEx, setSelEx] = useState(ALL_EX[0]);
  const [sets, setSets] = useState([{ reps: "", weight: "", rpe: "" }]);
  const [msg, setMsg] = useState("");
  const [histEx, setHistEx] = useState(ALL_EX[0]);
  const [chartMode, setChartMode] = useState("max");

  useEffect(() => {
    (async () => {
      try { const r = await window.storage.get("journal_v2"); if (r?.value) setSessions(JSON.parse(r.value)); } catch {}
      setLoaded(true);
    })();
  }, []);

  const persist = async (data) => { try { await window.storage.set("journal_v2", JSON.stringify(data)); } catch {} };
  const addSet = () => setSets([...sets, { reps: "", weight: "", rpe: "" }]);
  const rmSet = (i) => setSets(sets.filter((_, j) => j !== i));
  const upSet = (i, f, v) => setSets(sets.map((s, j) => j === i ? { ...s, [f]: v } : s));

  const logSession = async () => {
    const valid = sets.filter((s) => s.reps || s.weight);
    if (!valid.length) return;
    const entry = { id: Date.now(), date: new Date().toLocaleDateString("fr-FR"), dateISO: new Date().toISOString().split("T")[0], exercise: selEx, sets: valid };
    const updated = [...sessions, entry];
    setSessions(updated); await persist(updated);
    setSets([{ reps: "", weight: "", rpe: "" }]);
    setMsg("✓ Séance enregistrée !"); setTimeout(() => setMsg(""), 2500);
  };

// ── SUIVI TAB ─────────────────────────────────────────────────────────────────
function SuiviTab() {
  const [bodyData, setBodyData] = useState([]);
  const [recov, setRecov] = useState([]);
  const [loaded, setLoaded] = useState(false);
  const [bw, setBw] = useState("");
  const [sleep, setSleep] = useState("");
  const [fatigue, setFatigue] = useState(3);
  const [stress, setStress] = useState(3);
  const [msg, setMsg] = useState("");
  const [view, setView] = useState("form");

  useEffect(() => {
    (async () => {
      try {
        const b = await window.storage.get("body_tracking"); if (b?.value) setBodyData(JSON.parse(b.value));
        const r = await window.storage.get("recovery_tracking"); if (r?.value) setRecov(JSON.parse(r.value));
      } catch {}
      setLoaded(true);
    })();
  }, []);

  const save = async () => {
    const today = new Date().toLocaleDateString("fr-FR");
    const isoDate = new Date().toISOString().split("T")[0];
    let updatedB = bodyData, updatedR = recov;

    if (bw) {
      updatedB = [...bodyData, { date: today, dateISO: isoDate, weight: parseFloat(bw) }];
      setBodyData(updatedB);
      try { await window.storage.set("body_tracking", JSON.stringify(updatedB)); } catch {}
    }

    const entry = { id: Date.now(), date: today, dateISO: isoDate, sleep: parseFloat(sleep) || null, fatigue, stress };
    updatedR = [...recov, entry];
    setRecov(updatedR);
    try { await window.storage.set("recovery_tracking", JSON.stringify(updatedR)); } catch {}

    setBw(""); setSleep("");
    setMsg("✓ Données enregistrées !"); setTimeout(() => setMsg(""), 2500);
  };

  const bodyChart = bodyData.slice(-20).sort((a, b) => a.dateISO > b.dateISO ? 1 : -1);
  const recovChart = recov.slice(-14).sort((a, b) => a.dateISO > b.dateISO ? 1 : -1);

  const lastRecov = recov[recov.length - 1];
  let alertMsg = null;
  if (recovChart.length >= 3) {
    const lastThree = recovChart.slice(-3);
    const avgFatigue = lastThree.reduce((a, r) => a + r.fatigue, 0) / 3;
    const avgSleep = lastThree.filter((r) => r.sleep).reduce((a, r) => a + r.sleep, 0) / lastThree.filter((r) => r.sleep).length;
    if (avgFatigue >= 4) alertMsg = { type: "danger", msg: `Fatigue élevée depuis 3 jours (moy. ${avgFatigue.toFixed(1)}/5) → Réduis le volume de 20% cette semaine.` };
    else if (avgSleep && avgSleep < 6.5) alertMsg = { type: "warn", msg: `Sommeil insuffisant (moy. ${avgSleep.toFixed(1)}h) → Limite les séances à 4 cette semaine.` };
  }

  if (!loaded) return <div style={{ padding: 30, color: C.muted, textAlign: "center", fontSize: 12 }}>Chargement...</div>;

  return (
    <>
      <div style={{ display: "flex", gap: 6, marginBottom: 14 }}>
        {[{ id: "form", l: "📋 SAISIR" }, { id: "graphs", l: "📈 GRAPHIQUES" }].map((v) => (
          <button key={v.id} onClick={() => setView(v.id)} style={{ flex: 1, background: view === v.id ? C.accent : C.card, color: view === v.id ? C.dark : C.muted, border: `1px solid ${view === v.id ? C.accent : C.border}`, borderRadius: 8, padding: 10, fontSize: 11, fontWeight: 700, fontFamily: "'DM Mono',monospace", cursor: "pointer" }}>
            {v.l}
          </button>
        ))}
      </div>

      {alertMsg && (
        <div style={{ background: alertMsg.type === "danger" ? "#1A0000" : "#1A0E00", border: `1px solid ${alertMsg.type === "danger" ? C.red : C.yellow}55`, borderLeft: `3px solid ${alertMsg.type === "danger" ? C.red : C.yellow}`, borderRadius: 6, padding: "10px 12px", fontSize: 11, color: alertMsg.type === "danger" ? "#FCA5A5" : "#FCD34D", marginBottom: 12, lineHeight: 1.6 }}>
          {alertMsg.type === "danger" ? "🔴" : "⚠️"} {alertMsg.msg}
        </div>
      )}

      {view === "form" && (
        <div style={s.card}>
          <div style={s.ct}>SAISIE QUOTIDIENNE — {new Date().toLocaleDateString("fr-FR")}</div>

          <div style={{ fontSize: 11, fontWeight: 700, color: C.blue, letterSpacing: 1, marginBottom: 8 }}>POIDS DE CORPS (kg)</div>
          <input type="number" min="50" max="200" step="0.1" placeholder="ex : 89.4" value={bw} onChange={(e) => setBw(e.target.value)} style={{ ...s.inp, marginBottom: 14, color: C.blue, fontSize: 16 }} />

          <div style={{ fontSize: 11, fontWeight: 700, color: C.blue, letterSpacing: 1, marginBottom: 8 }}>SOMMEIL (heures)</div>
          <input type="number" min="3" max="12" step="0.5" placeholder="ex : 7.5" value={sleep} onChange={(e) => setSleep(e.target.value)} style={{ ...s.inp, marginBottom: 14, color: C.blue }} />

          <div style={{ fontSize: 11, fontWeight: 700, color: C.yellow, letterSpacing: 1, marginBottom: 8 }}>FATIGUE MUSCULAIRE — {fatigue}/5</div>
          <div style={{ display: "flex", gap: 6, marginBottom: 14 }}>
            {[1, 2, 3, 4, 5].map((v) => (
              <button key={v} onClick={() => setFatigue(v)} style={{ flex: 1, background: fatigue >= v ? C.yellow : "#1E2330", color: fatigue >= v ? C.dark : C.muted, border: "none", borderRadius: 6, padding: "8px 0", fontSize: 12, fontWeight: 700, fontFamily: "'DM Mono',monospace", cursor: "pointer" }}>{v}</button>
            ))}
          </div>

          <div style={{ fontSize: 11, fontWeight: 700, color: C.pink, letterSpacing: 1, marginBottom: 8 }}>STRESS GÉNÉRAL — {stress}/5</div>
          <div style={{ display: "flex", gap: 6, marginBottom: 16 }}>
            {[1, 2, 3, 4, 5].map((v) => (
              <button key={v} onClick={() => setStress(v)} style={{ flex: 1, background: stress >= v ? C.pink : "#1E2330", color: stress >= v ? C.dark : C.muted, border: "none", borderRadius: 6, padding: "8px 0", fontSize: 12, fontWeight: 700, fontFamily: "'DM Mono',monospace", cursor: "pointer" }}>{v}</button>
            ))}
          </div>

          <button onClick={save} style={{ ...s.btn(), width: "100%", padding: 12 }}>ENREGISTRER</button>
          {msg && <div style={{ fontSize: 12, color: C.accent, textAlign: "center", marginTop: 10 }}>{msg}</div>}
        </div>
      )}

      {view === "graphs" && (
        <>
          {bodyChart.length >= 2 ? (
            <div style={s.card}>
              <div style={s.ct}>POIDS DE CORPS (kg)</div>
              <ResponsiveContainer width="100%" height={150}>
                <LineChart data={bodyChart} margin={{ top: 4, right: 8, left: -24, bottom: 0 }}>
                  <CartesianGrid strokeDasharray="3 3" stroke="#1E2330" />
                  <XAxis dataKey="date" tick={{ fill: C.muted, fontSize: 8 }} tickLine={false} axisLine={false} />
                  <YAxis tick={{ fill: C.muted, fontSize: 8 }} tickLine={false} axisLine={false} domain={["auto", "auto"]} />
                  <Tooltip contentStyle={{ background: "#141720", border: `1px solid ${C.border}`, borderRadius: 6, fontSize: 11, fontFamily: "'DM Mono',monospace" }} labelStyle={{ color: C.blue }} />
                  <Line type="monotone" dataKey="weight" stroke={C.blue} strokeWidth={2} dot={{ fill: C.blue, r: 4, strokeWidth: 0 }} name="Poids (kg)" />
                </LineChart>
              </ResponsiveContainer>
            </div>
          ) : <div style={{ ...s.card, textAlign: "center", color: C.muted, fontSize: 11, padding: "20px 16px" }}>Enregistre au moins 2 jours pour voir le graphique poids.</div>}

          {recovChart.length >= 2 ? (
            <div style={s.card}>
              <div style={s.ct}>FATIGUE & SOMMEIL</div>
              <ResponsiveContainer width="100%" height={150}>
                <LineChart data={recovChart} margin={{ top: 4, right: 8, left: -24, bottom: 0 }}>
                  <CartesianGrid strokeDasharray="3 3" stroke="#1E2330" />
                  <XAxis dataKey="date" tick={{ fill: C.muted, fontSize: 8 }} tickLine={false} axisLine={false} />
                  <YAxis tick={{ fill: C.muted, fontSize: 8 }} tickLine={false} axisLine={false} />
                  <Tooltip contentStyle={{ background: "#141720", border: `1px solid ${C.border}`, borderRadius: 6, fontSize: 11, fontFamily: "'DM Mono',monospace" }} labelStyle={{ color: C.yellow }} />
                  <Line type="monotone" dataKey="fatigue" stroke={C.yellow} strokeWidth={2} dot={{ fill: C.yellow, r: 3, strokeWidth: 0 }} name="Fatigue /5" />

                </LineChart>
              </ResponsiveContainer>
              <div style={{ display: "flex", gap: 12, marginTop: 8 }}>

                <span style={{ fontSize: 10, color: C.blue }}>■ Sommeil (h)</span>

            </div>
          ) : <div style={{ ...s.card, textAlign: "center", color: C.muted, fontSize: 11, padding: "20px 16px" }}>Enregistre au moins 2 jours pour voir l'évolution.</div>}
        </>
      )}
    </>
  );
